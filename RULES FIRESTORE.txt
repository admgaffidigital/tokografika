rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // 1. ATURAN LISENSI (Kunci Rapat 🔒)
    // Hanya izinkan aplikasi mencocokkan 1 kode spesifik. Dilarang melihat semua daftar.
    match /freshmart_licenses/{licenseId} {
      allow get: if true; 
      allow list, write: if false;
    }
    
    // 2. DATA TOKO & PRODUK (Buka Akses 🔓)
    match /freshmart/{document=**} {
      allow read, write: if true; 
    }

    // 3. DATA PESANAN (Buka Akses 🔓)
    match /freshmart_orders/{document=**} {
      allow read, write: if true; 
    }

    // 4. DATA PENARIKAN DANA DRIVER (Buka Akses 🔓)
    match /freshmart_withdrawals/{document=**} {
      allow read, write: if true; 
    }
  }
}
